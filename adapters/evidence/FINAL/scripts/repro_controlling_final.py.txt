def report(finding: str, reproduced, detail: str) -> None:
    """``reproduced`` is True / False, or None for a probe that could not run (never counted as NOT REPRODUCED)."""
    RESULTS.append((finding, reproduced, detail))
    label = "PROBE ERROR" if reproduced is None else ("REPRODUCED" if reproduced else "NOT REPRODUCED")
    print(f"{finding:<34} {label:<15} {detail}", flush=True)


def probe(finding: str):
    def wrap(function):
        try:
            function()
        except Exception as exc:                                    # a probe that cannot run is reported, not hidden
            text = "".join(traceback.format_exception_only(type(exc), exc)).strip().splitlines()[-1]
            if SCANNER.scan(text.encode()).hit:
                text = type(exc).__name__ + " <message withheld>"
            report(finding, None, f"PROBE ERROR: {text[:150]}")
        return function
    return wrap


def runtime_hits(root: Path, secret: Secret) -> int:
    return len(verify.scan_runtime_for_secret(root, secret))


class Clock:
    def __init__(self, start="2026-10-01T12:00:00.000000Z", step=1000):
        self.inner = FixedClock(start, step_micros=step)

    def now(self):
        return self.inner.now()

# -- C:HA-02 ------------------------------------------------------------------------------------------
@probe("C:HA-02 close() process control")
def _():
    outcomes = []
    for injected in (KeyboardInterrupt("close-error: " + SENTINEL_KEY), SystemExit("close-error: " + SENTINEL_KEY)):
        class Conn:
            def __init__(self, *args):
                self.done = False

            def open(self, timeout):
                pass

            def settimeout(self, timeout):
                pass

            def write(self, method, target, headers):
                pass

            def read_head(self):
                return 200, [("Content-Type", "application/json")]

            def read_chunk(self, size):
                if self.done:
                    return b""
                self.done = True
                return b"[]"

            def finished(self):
                return self.done

            def close(self, exc=injected):
                raise exc

        transport = th.HttpsTransport(Secret(SENTINEL_KEY), credential_param="apiKey", policy=ps.POLICY,
                                      connection_factory=Conn)
        try:
            transport.send(request(), clock=Clock(), deadline_at="2026-10-01T13:00:00.000000Z")
            outcomes.append((type(injected).__name__, "swallowed", False))
        except BaseException as caught:                             # noqa: BLE001 - the probe inspects it
            same_object = caught is injected
            text = SCANNER.scan(repr(caught.args).encode()).hit
            keyed_context = caught.__context__ is not None or caught.__cause__ is not None
            notes = hasattr(caught, "__notes__")
            outcomes.append((type(caught).__name__, "same" if same_object else "fresh",
                             same_object or text or keyed_context or notes))
    report("C:HA-02 close() process control", any(bad for _, _, bad in outcomes),
           "; ".join(f"{cls}:{kind}:unsafe={bad}" for cls, kind, bad in outcomes))

@probe("C:HA-03 write at deadline")
def _():
    before, deadline = "2026-10-01T12:00:00.000000Z", "2026-10-01T12:00:01.000000Z"
    state = {"value": before}

    class DeadlineClock:
        def now(self):
            return state["value"]

    writes = []

    class Conn:
        def __init__(self, *args):
            pass

        def open(self, timeout):
            pass

        def settimeout(self, timeout):
            state["value"] = deadline                                  # the deadline arrives before the write

        def write(self, *args):
            writes.append(state["value"])
            raise OSError("stop after write")

        def close(self):
            pass

    transport = th.HttpsTransport(Secret("SENTINEL-12345678"), credential_param="apiKey", policy=ps.POLICY,
                                  connection_factory=Conn)
    result = transport.send(request(), clock=DeadlineClock(), deadline_at=deadline)
    report("C:HA-03 write at deadline", bool(writes),
           f"writes at/after the deadline={len(writes)}; outcome={result.outcome}")

def _():
    key = "éèêëàáâã"
    mojibake = key.encode("utf-8").decode("latin-1")                   # what http.client hands over
    with scratch_root() as root:

        hits = runtime_hits(root, Secret(key))
