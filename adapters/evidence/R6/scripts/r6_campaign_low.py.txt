def run(label, rel, old, new, tests):
    (SCR / "camp_old.txt").write_text(old)
    (SCR / "camp_new.txt").write_text(new)
    proc = subprocess.run(["python3", str(SCR / "mutate_run.py"), label, rel, "@" + str(SCR / "camp_old.txt"),
                           "@" + str(SCR / "camp_new.txt"), *tests], capture_output=True, text=True)
    return (proc.stdout.strip() or proc.stderr.strip()[-300:]).replace("\n", " ")


def main():
    bad = 0
    for label, rel, old, new, tests in M:
        text = pathlib.Path(rel).read_text() if pathlib.Path(rel).exists() else ""
        if text.count(old) != 1 or old == new:
            bad += 1
            print("PROBLEM", label, text.count(old))
    if bad:
        sys.exit(f"{bad} mutant specifications do not match the source exactly once")
    only = sys.argv[1:] or None
    with open(OUT, "a") as out:
        for label, rel, old, new, tests in M:
            if only and not any(label.startswith(prefix) for prefix in only):
                continue
            out.write(run(label, rel, old, new, tests) + "\n")
            out.flush()
        out.write("DONE\n")


main()
