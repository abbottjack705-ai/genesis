import json, socket, subprocess, sys, os
out = {}
def attempt(label, fn):
    try:
        fn(); out[label] = "ALLOWED"
    except PermissionError: out[label] = "BLOCKED"
    except OSError as e: out[label] = "ALLOWED(" + type(e).__name__ + ")"
attempt("connect 192.0.2.1:443", lambda: socket.create_connection(("192.0.2.1", 443), timeout=2))
attempt("getaddrinfo example.com", lambda: socket.getaddrinfo("example.com", 443))
attempt("gethostbyname api.oddspapi.io", lambda: socket.gethostbyname("api.oddspapi.io"))
attempt("connect 127.0.0.1:1", lambda: socket.create_connection(("127.0.0.1", 1), timeout=2))
attempt("connect localhost:1", lambda: socket.create_connection(("localhost", 1), timeout=2))
if len(sys.argv) < 2:
    child = subprocess.run([sys.executable, "-B", __file__, "child"], capture_output=True, text=True)
    out["child"] = json.loads(child.stdout)
print(json.dumps(out))
